2//! JWKS helper (H13) — JWKS URL builder + fetch + cache.
3
4use std::sync::Arc;
5use std::time::{Duration, Instant};
6
7use http::{HeaderMap, Method};
8use jsonwebtoken::jwk::JwkSet;
9use tokio::sync::RwLock;
10
11use crate::client::{Client, DEFAULT_BASE_URL};
12use crate::error::Error;
13use crate::transport::{HttpRequest, SharedTransport};
14
15/// Builds the JWKS URL for `client_id` against `base_url` (defaulting when empty).
16pub fn jwks_url(base_url: &str, client_id: &str) -> String {
17    let base = if base_url.is_empty() {
18        DEFAULT_BASE_URL
19    } else {
20        base_url
21    };
22    format!("{base}/sso/jwks/{client_id}")
23}
24
25#[derive(Clone)]
26struct Cached {
27    set: Arc<JwkSet>,
28    fetched_at: Instant,
29}
30
31/// Caches the JWKS for a given client. Default TTL 10 minutes.
32pub struct JwksHelper {
33    transport: SharedTransport,
34    url: String,
35    ttl: Duration,
36    cache: RwLock<Option<Cached>>,
37}
38
39impl JwksHelper {
40    /// Build a JWKS helper that uses the supplied transport.
41    pub fn with_transport(
42        transport: SharedTransport,
43        base_url: impl AsRef<str>,
44        client_id: impl AsRef<str>,
45    ) -> Self {
46        Self {
47            transport,
48            url: jwks_url(base_url.as_ref(), client_id.as_ref()),
49            ttl: Duration::from_secs(600),
50            cache: RwLock::new(None),
51        }
52    }
53
54    /// Construct a JWKS helper from a client — reuses the client's transport.
55    pub fn from_client(client: &Client) -> Self {
56        Self::with_transport(client.transport(), client.base_url(), client.client_id())
57    }
58
59    pub fn with_ttl(mut self, ttl: Duration) -> Self {
60        self.ttl = ttl;
61        self
62    }
63
64    /// Returns the JWKS URL.
65    pub fn url(&self) -> &str {
66        &self.url
67    }
68
69    /// Fetches (and caches) the JWKS. A subsequent call within `ttl` returns
70    /// the cached set without making a network request.
71    pub async fn fetch(&self) -> Result<Arc<JwkSet>, Error> {
72        if let Some(c) = self.cache.read().await.as_ref()
73            && c.fetched_at.elapsed() < self.ttl
74        {
75            return Ok(c.set.clone());
76        }
77        let req = HttpRequest {
78            method: Method::GET,
79            url: self.url.clone(),
80            headers: HeaderMap::new(),
81            body: None,
82        };
83        let resp = self.transport.execute(req).await.map_err(Error::Network)?;
84        if !resp.status.is_success() {
85            return Err(Error::Api {
86                status: resp.status.as_u16(),
87                code: None,
88                message: String::from_utf8_lossy(&resp.body).into_owned(),
89            });
90        }
91        let set: JwkSet = serde_json::from_slice(&resp.body).map_err(Error::from)?;
92        let arc = Arc::new(set);
93        let mut guard = self.cache.write().await;
94        *guard = Some(Cached {
95            set: arc.clone(),
96            fetched_at: Instant::now(),
97        });
98        Ok(arc)
99    }
100
101    /// Force-refresh the cache.
102    pub async fn refresh(&self) -> Result<Arc<JwkSet>, Error> {
103        {
104            let mut guard = self.cache.write().await;
105            *guard = None;
106        }
107        self.fetch().await
108    }
109}
